# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.30.0
engine.minor=32
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The mine is also played with the pointer or touch: tap a layer (or its
# label) to select it, tap a flaring layer to claim the flare, drag the mine
# to turn it a little; the Foreman's Panel, the Shift Report and the phone's
# action tray hold every command as buttons.
control.up=ArrowUp | Select layer above
control.down=ArrowDown | Select layer below
control.assign=Enter | Assign worker
control.unassign=KeyU | Recall worker
control.hire=KeyH | Hire
control.shaft=KeyQ | Upgrade shaft
control.liftcap=KeyW | Upgrade lift capacity
control.liftspeed=KeyE | Upgrade lift speed
control.unlock=KeyD | Open next layer
control.flare=KeyF | Claim flare
control.foreman=KeyG | Foreman buy / toggle
control.undo=KeyZ | Undo (practice)
control.pause=Escape | Pause / back
engine.group.up=Mine
engine.group.hire=Upgrades
engine.group.flare=Shift

# Gamepad defaults (the browser build's: D-pad selects, A assigns or claims
# the selected layer's flare, B recalls, Start pauses; the other commands
# gained buttons of their own).
engine.gamepad.up=DpadUp+LeftStickUp
engine.gamepad.down=DpadDown+LeftStickDown
engine.gamepad.assign=South
engine.gamepad.unassign=East
engine.gamepad.hire=North
engine.gamepad.shaft=RightShoulder
engine.gamepad.liftcap=LeftTrigger
engine.gamepad.liftspeed=RightTrigger
engine.gamepad.unlock=LeftShoulder
engine.gamepad.flare=West
engine.gamepad.foreman=Select
engine.gamepad.undo=LeftStickPress
engine.gamepad.pause=Start
